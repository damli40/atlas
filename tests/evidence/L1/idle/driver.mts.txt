// The real MCP program over stdio, a whole round, then idle, then the person's choice. Models mocked: $0.
import { spawn } from "node:child_process";
import { appendFileSync, createWriteStream } from "node:fs";
const dir = process.argv[2]!; const idleMs = Number(process.argv[3]);
const log = (s: string) => appendFileSync(`${dir}/run.log`, `${new Date().toISOString()} ${s}\n`);
const child = spawn("node", ["--experimental-strip-types", "--no-warnings", `${dir}/plugin/mcp/main.ts`], { cwd: `${dir}/ws`, env: { PATH: process.env.PATH!, HOME: `${dir}/data`, ATLAS_URL: "http://127.0.0.1:8798", MUSE_PLUGIN_DATA_DIR: `${dir}/data` }, stdio: ["pipe", "pipe", "pipe"] });
child.stderr.pipe(createWriteStream(`${dir}/stderr.txt`));
child.on("exit", (code, signal) => { log(`EXIT child code=${code} signal=${signal}`); process.exit(1); });
let buf = ""; let id = 0; const wait = new Map<number, (v: any) => void>();
child.stdout.setEncoding("utf8");
child.stdout.on("data", (d: string) => { buf += d; let n; while ((n = buf.indexOf("\n")) >= 0) { const m = JSON.parse(buf.slice(0, n)); buf = buf.slice(n + 1); wait.get(m.id)?.(m); } });
const rpc = (method: string, params: unknown) => new Promise<any>((ok) => { const i = ++id; wait.set(i, ok); child.stdin.write(JSON.stringify({ jsonrpc: "2.0", id: i, method, params }) + "\n"); });
const tool = async (name: string, args: unknown) => { const r = await rpc("tools/call", { name, arguments: args }); const t: string = r.result.content[0].text; log(`${name}: ${r.result.isError ? "ERROR " : ""}${t.split("\n").slice(0, 2).join(" | ").slice(0, 200)}`); return t; };
const pause = (ms: number) => new Promise((r) => setTimeout(r, ms));
await rpc("initialize", { protocolVersion: "2025-06-18", capabilities: {}, clientInfo: { name: "idle", version: "0" } });
await tool("atlas_read_file_function", { path: "retry.ts" });
const props = await tool("atlas_propose_tests", { path: "retry.ts", function: "retryWithBackoff" });
const ids = [...props.matchAll(/\[([\w-]+)\] allowed/g)].map((m) => m[1]!);
const start = await tool("atlas_start_round", { path: "retry.ts", proposal_id: ids[1] });
const rid = /Round (\S+) is open/.exec(start)![1]!;
await pause(2300);
await tool("atlas_seal_bet", { experiment_id: rid, answers: [{ measures: "outcome", typed: "threw" }, { measures: "virtualMs", typed: "700" }, { measures: "calls:fetchIt", typed: "3" }] });
await tool("atlas_run", { experiment_id: rid });
log(`IDLE start pid=${child.pid} for ${idleMs / 60000} min`);
await pause(idleMs);
log(`IDLE over; child alive=${child.exitCode === null}`);
const c = await tool("atlas_choose_target", { experiment_id: rid, typed: "my answers are what I want" });
await tool("atlas_status", {});
log(`RESULT choose ok=${c.includes("Their sealed answers are now the target")}`);
child.kill(); process.exit(0);

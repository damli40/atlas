// L1 fallback after Muse hit the $2 stop: the REAL plugin MCP program over stdio, the real local Atlas server, real Muse
// Wonder + Muse Spark. This script stands where Muse Code stands: it calls the tools and writes the edits.
import { spawn } from "node:child_process";
import { readFileSync, writeFileSync } from "node:fs";
import { createServer, request } from "node:http";
import { verifyAtlasFile } from "<repo>/src/atlasfile.ts";
const SCR = process.argv[2]!;
const PLUGIN = "<repo>/plugins/muse-atlas/mcp/main.ts";
const invite = readFileSync(`${SCR}/invite`, "utf8").trim();
const out = `${SCR}/stdio`;
let session = "";
const proxy = createServer((req, res) => {
  const chunks: Buffer[] = [];
  req.on("data", (c) => chunks.push(c));
  req.on("end", () => {
    session = String(req.headers["x-atlas-session"] ?? session);
    const up = request({ host: "127.0.0.1", port: 8797, path: req.url, method: req.method, headers: req.headers }, (r) => { res.writeHead(r.statusCode ?? 500, r.headers); r.pipe(res); });
    up.end(Buffer.concat(chunks));
  });
});
await new Promise<void>((ok) => proxy.listen(0, "127.0.0.1", ok));
const port = (proxy.address() as { port: number }).port;
const child = spawn("node", ["--experimental-strip-types", "--no-warnings", PLUGIN], { cwd: `${SCR}/ws2`, env: { PATH: process.env.PATH!, HOME: `${SCR}/home`, ATLAS_URL: `http://127.0.0.1:${port}`, ATLAS_INVITE: invite }, stdio: ["pipe", "pipe", "inherit"] });
let buf = ""; let id = 0;
const wait = new Map<number, (v: any) => void>();
child.stdout.setEncoding("utf8");
child.stdout.on("data", (d: string) => { buf += d; let n; while ((n = buf.indexOf("\n")) >= 0) { const m = JSON.parse(buf.slice(0, n)); buf = buf.slice(n + 1); wait.get(m.id)?.(m); } });
const rpc = (method: string, params: unknown) => new Promise<any>((ok) => { const i = ++id; wait.set(i, ok); child.stdin.write(JSON.stringify({ jsonrpc: "2.0", id: i, method, params }) + "\n"); });
let step = 0;
async function tool(name: string, args: unknown): Promise<string> {
  const r = await rpc("tools/call", { name, arguments: args });
  const text: string = r.result.content[0].text;
  step++;
  writeFileSync(`${out}/${String(step).padStart(2, "0")}-${name}.txt`, `> ${name} ${JSON.stringify(args)}\n${r.result.isError ? "[isError]\n" : ""}${text}\n`);
  return text;
}
const pause = (ms: number) => new Promise((r) => setTimeout(r, ms));
await rpc("initialize", { protocolVersion: "2025-06-18", capabilities: {}, clientInfo: { name: "l1-driver", version: "0" } });
await tool("atlas_read_file_function", { path: "retry.ts" });
const props = await tool("atlas_propose_tests", { path: "retry.ts", function: "retryWithBackoff" });
const pick = [...props.matchAll(/\[([\w-]+)\] allowed: ([^\n]*)/g)];
const always = pick.find((m) => !/return/i.test(m[2]!)) ?? pick[0]!;
const start = await tool("atlas_start_round", { path: "retry.ts", proposal_id: always[1] });
const rid = /Round (\S+) is open/.exec(start)![1]!;
await pause(2300);
await tool("atlas_seal_bet", { experiment_id: rid, answers: [{ measures: "outcome", typed: "threw" }, { measures: "virtualMs", typed: "700" }, { measures: "calls:fetchIt", typed: "3" }] });
const ran = await tool("atlas_run", { experiment_id: rid });
const refused = await tool("atlas_rerun", { experiment_id: rid }); // before any choice
void refused; void ran;
await pause(2300);
await tool("atlas_choose_target", { experiment_id: rid, typed: "yes, go ahead" }); // not the words
await tool("atlas_choose_target", { experiment_id: rid, typed: "my answers are what I want" });
await tool("atlas_rerun", { experiment_id: rid }); // no edit yet: no try used
const src = readFileSync(`${SCR}/ws2/retry.ts`, "utf8");
writeFileSync(`${SCR}/ws2/retry.ts`, src.replace("setTimeout(r, baseMs)", "setTimeout(r, baseMs * 2)"));
await tool("atlas_rerun", { experiment_id: rid });
writeFileSync(`${SCR}/ws2/retry.ts`, src.replace("setTimeout(r, baseMs)", "setTimeout(r, baseMs * 2 ** i)"));
const last = await tool("atlas_rerun", { experiment_id: rid });
const runId = /proofs\/([\w-]+)/.exec(last)?.[1] ?? /proof=([\w-]+)/.exec(last)![1]!;
const res = await fetch(`http://127.0.0.1:8797/proofs/${runId}`, { headers: { "x-atlas-session": session, "x-atlas-invite": invite } });
const file = await res.json();
writeFileSync(`${out}/proof-${runId}.atlas.json`, JSON.stringify(file, null, 2));
const v = verifyAtlasFile(file);
const tampered = JSON.parse(JSON.stringify(file)); tampered.loop.tries[1].verdicts[0].outcome = tampered.loop.tries[1].verdicts[0].outcome === "matched" ? "not_matched" : "matched";
const t = verifyAtlasFile(tampered);
const summary = `verify: ${v.ok ? "ok" : (v as any).problems.join("; ")}\ntampered copy: ${t.ok ? "OK (BAD)" : (t as any).problems.join("; ").slice(0, 300)}\nmeasured per try: ${JSON.stringify(file.loop.tries.map((x: any) => x.run.result.measured.virtualMs))}\nseeds: ${JSON.stringify([file.runs[0].seed, ...file.loop.tries.map((x: any) => x.run.seed)])}\nstopped: ${file.loop.stopped?.reason}\n`;
writeFileSync(`${out}/verify.txt`, summary);
console.log(summary);
child.kill(); proxy.close(); process.exit(0);

128|     function _deployCoreWrappers(Addresses addresses) internal {
129|         OracleConfig[] memory oracleConfigs = getOracleConfigurations(
130|             block.chainid
131|         );
132| 
133|         if (oracleConfigs.length == 0) {
134|             console.log("No oracle configs found for chain %d", block.chainid);
135|             return;
136|         }
137| 
138|         vm.startBroadcast();
139| 
140|         for (uint256 i = 0; i < oracleConfigs.length; i++) {
141|             OracleConfig memory config = oracleConfigs[i];
142| 
143|             string memory wrapperName = string(
144|                 abi.encodePacked(config.oracleName, "_OEV_WRAPPER")
145|             );
146| 
147|             // If a wrapper already exists for this oracle, check if it needs
148|             // deprecation or if it was already handled by a previous config
149|             // (e.g. USDC/USDBC both reference CHAINLINK_USDC_USD)
150|             if (addresses.isAddressSet(wrapperName)) {
151|                 string memory deprecatedName = string(
152|                     abi.encodePacked(wrapperName, "_DEPRECATED")
153|                 );
154| 
155|                 // Skip if already deprecated by a previous iteration
156|                 if (addresses.isAddressSet(deprecatedName)) {
157|                     continue;
158|                 }
159| 
160|                 // Skip if the existing wrapper is a new-style one deployed by
161|                 // a previous iteration in this loop. Old-style wrappers
162|                 // (ChainlinkFeedOEVWrapper) lack the chainlinkOracle() field.
163|                 try
164|                     ChainlinkOEVWrapper(
165|                         payable(addresses.getAddress(wrapperName))
166|                     ).chainlinkOracle()
167|                 returns (IChainlinkOracle) {
168|                     continue;
169|                 } catch {}
170| 
171|                 address oldWrapper = addresses.getAddress(wrapperName);
172|                 addresses.addAddress(deprecatedName, oldWrapper);
173| 
174|                 ChainlinkOEVWrapper wrapper = new ChainlinkOEVWrapper(
175|                     addresses.getAddress(config.oracleName),
176|                     addresses.getAddress("TEMPORAL_GOVERNOR"),
177|                     addresses.getAddress("CHAINLINK_ORACLE"),
178|                     addresses.getAddress("OEV_PROTOCOL_FEE_REDEEMER"),
179|                     FEE_MULTIPLIER,
180|                     MAX_ROUND_DELAY,
181|                     MAX_DECREMENTS
182|                 );
183| 
184|                 addresses.changeAddress(wrapperName, address(wrapper), true);
185|             } else {
186|                 ChainlinkOEVWrapper wrapper = new ChainlinkOEVWrapper(
187|                     addresses.getAddress(config.oracleName),
188|                     addresses.getAddress("TEMPORAL_GOVERNOR"),
189|                     addresses.getAddress("CHAINLINK_ORACLE"),
190|                     addresses.getAddress("OEV_PROTOCOL_FEE_REDEEMER"),
191|                     FEE_MULTIPLIER,
192|                     MAX_ROUND_DELAY,
193|                     MAX_DECREMENTS
194|                 );
195| 
196|                 addresses.addAddress(wrapperName, address(wrapper));
197|             }
198|         }
199| 
200|         vm.stopBroadcast();
201|     }

  1| # MIP-X43: Activate OEV Wrappers for Non-Composite Markets
  2| 
  3| ## Summary
  4| 
  5| This proposal activates Chainlink OEV Wrapper contracts across all remaining
  6| Moonwell non-composite oracle markets on Base and Optimism. Following successful
  7| validation of OEV functionality in MIP-X38, this proposal expands coverage to 21
  8| additional core markets and 2 Morpho isolated markets.
  9| 
 10| Composite oracle markets are excluded from this proposal and will be addressed
 11| in a subsequent proposal once compatibility is implemented.
 12| 
 13| Additionally, this proposal updates the OEV fee split on existing and newly
 14| deployed wrappers from **60/40 to 70/30 (protocol/liquidator)** to improve
 15| protocol value capture based on performance observed during the initial rollout.
 16| 
 17| ## Background
 18| 
 19| MIP-X38 deployed the updated OEV wrapper infrastructure and activated it on
 20| select feeds as a production validation step:
 21| 
 22| - Base Core — WETH (CHAINLINK_ETH_USD)
 23| - Base Morpho — WELL (CHAINLINK_WELL_USD)
 24| - Optimism Core — WETH (CHAINLINK_ETH_USD)
 25| 
 26| With the infrastructure validated, the protocol can extend wrapper activation to
 27| remaining markets utilizing standard Chainlink price feeds.
 28| 
 29| Markets using composite oracle structures are excluded because their oracle
 30| implementations are not yet compatible with the current wrapper architecture.
 31| These markets will be addressed in a future proposal once compatibility support
 32| is introduced.
 33| 
 34| Data gathered from the initial deployment phase also indicated that the protocol
 35| can sustainably increase its share of OEV revenue. This proposal therefore
 36| includes an update to the fee split to better align incentives and protocol
 37| revenue capture.
 38| 
 39| ## Proposal Details
 40| 
 41| ### Core Markets
 42| 
 43| - Deploy new **ChainlinkOEVWrapper** contracts for:
 44|   - 14 Base markets
 45|   - 7 Optimism markets
 46| - Replace legacy wrappers where applicable
 47| - Configure each wrapper as the oracle feed via `setFeed()` on ChainlinkOracle
 48| - Whitelist corresponding mTokens on OEVProtocolFeeRedeemer
 49| 
 50| **Base Core Markets**
 51| 
 52| DAI, USDC, USDbC, cbETH, AERO, cbBTC, EURC, WELL, USDS, tBTC, VIRTUAL, MORPHO,
 53| cbXRP, MAMO
 54| 
 55| **Optimism Core Markets**
 56| 
 57| USDC, USDT, DAI, WBTC, OP, VELO, USDT0
 58| 
 59| ### Isolated Markets
 60| 
 61| Upgrade the following oracle proxies to the **ChainlinkOEVMorphoWrapper**
 62| implementation:
 63| 
 64| - CHAINLINK_MAMO_USD_ORACLE_PROXY
 65| - CHAINLINK_stkWELL_USD_ORACLE_PROXY
 66| 
 67| Execution method:
 68| 
 69| - `upgradeAndCall()`
 70| - `initializeV2()`
 71| 
 72| ### OEV Fee Split Update
 73| 
 74| Update the OEV fee split from **60/40 to 70/30 (protocol/liquidator)** by
 75| modifying the `liquidatorFeeBps` parameter from **4000 → 3000** on existing
 76| wrappers deployed in MIP-X38:
 77| 
 78| - Base — CHAINLINK_ETH_USD_OEV_WRAPPER
 79| - Base Morpho — CHAINLINK_WELL_USD_ORACLE_PROXY
 80| - Optimism — CHAINLINK_ETH_USD_OEV_WRAPPER
 81| 
 82| All newly deployed wrappers in this proposal will use the updated **70/30
 83| split** at deployment.
 84| 
 85| ## Rationale
 86| 
 87| Expanding OEV wrapper coverage increases the protocol’s ability to capture
 88| oracle extractable value generated during liquidation events across compatible
 89| markets, strengthening reserve growth and standardizing oracle infrastructure
 90| across deployments.
 91| 
 92| Excluding composite oracle markets ensures deployment safety while compatibility
 93| work continues, preventing unnecessary integration risk.
 94| 
 95| Updating the fee split improves protocol revenue capture while maintaining
 96| incentives for liquidators, based on data from the initial rollout phase.
 97|

101|   specified
